The ladders (step i = 2*i on the glm scale; every question is prefixed with the `PRE` text):

```python
# Six concrete steps per axis; step i reads as 2*i on the glm 0-10 scale.
PRE = "Judge only the headline and the text. 0 means the trait is absent or does not apply; that is not a penalty. "
L = {
 "positivity": ("How much does the story as a whole leave the reader feeling good?", [
   "gloomy, or neutral information with no emotion", "mostly neutral with a faint good note",
   "mildly pleasant", "clearly good news, the reader is glad", "very uplifting, warm joy", "pure joy, nothing spoils it"]),
 "negativity": ("How much suffering, damage, loss or conflict is in the content, including the backstory? Not the mirror of positivity: a rescue from a fire can be both.", [
   "no negativity at all", "a passing mention of a difficulty", "a real problem in the backstory",
   "clear hardship: illness, accident, loss, conflict", "serious suffering or danger fills much of the story", "heavy: death, disaster, violence at the centre"]),
 "heartwarming": ("Warmth of heart: kindness, care, mutual help between people or towards animals.", [
   "no human warmth, a dry fact", "a hint of kindness", "some care or kindness is part of the story",
   "kindness or care is a main thread", "a touching act that warms the heart", "moving to tears"]),
 "cuteness": ("Cuteness: animals, baby animals, small children, cosy details.", [
   "nothing cute", "a cute detail in passing", "somewhat cute: an animal or a child appears",
   "clearly cute: animals or children are central", "very cute: babies, cubs, funny little habits", "maximally cute, 'aww' from start to end"]),
 "humor": ("Funniness: makes you smile and want to retell it as a joke.", [
   "not funny at all", "a light smile at one detail", "a bit amusing",
   "clearly funny, a quirky situation", "very funny, you want to retell it", "hilarious, laugh out loud"]),
 "pride_humanity": ("Pride in humanity: achievements of science and progress, human solidarity.", [
   "no reason for pride", "a small personal success", "a solid achievement of one person or team",
   "a notable achievement or a strong act of solidarity", "a major breakthrough or a feat of many people", "a triumph of humanity"]),
 "pride_russia": ("Pride in Russia: Russian achievements, people, cities and regions. 0 when the story is not about Russia.", [
   "not about Russia, or no reason", "Russia is mentioned in passing", "a small local Russian success",
   "a clear achievement of Russian people or a region", "a major Russian achievement", "unconditional national pride"]),
 "heroism": ("Heroism: a heroic deed, self-sacrifice, saving a life at risk to oneself.", [
   "no heroics", "courage in everyday life", "a brave act without real risk",
   "a rescue or a risky act of help", "a rescue at serious risk to the rescuer", "a true heroic deed, self-sacrifice"]),
 "inspiration": ("Inspiring force: does it motivate the reader to act, 'I can / want to do that too'?", [
   "does not motivate", "a faint 'nice for them'", "an example some readers could follow",
   "clearly inspiring, a model to follow", "strongly inspiring, hard to stay indifferent", "makes you want to act right now"]),
 "beauty": ("Aesthetics: beauty of a place, event or object as the text describes it.", [
   "nothing beautiful", "a pleasant image in passing", "some visual beauty is described",
   "beauty is a clear part of the story", "a striking, beautiful sight is central", "breathtakingly beautiful"]),
 "interestingness": ("How much does the story grip and hold attention?", [
   "boring: routine, a list, an announcement", "of niche interest only", "mildly interesting, read once",
   "interesting, most readers finish it", "gripping, you tell a friend", "can't stop reading, a story everyone talks about"]),
 "surprise": ("Does the fact or the outcome surprise the reader?", [
   "entirely predictable", "a small unexpected detail", "a somewhat unexpected fact",
   "clearly surprising", "a very surprising fact or twist", "a complete surprise, 'really?!'"]),
 "uniqueness": ("Rarity or strangeness of the phenomenon itself: 'this doesn't happen'.", [
   "an ordinary thing that happens every day", "slightly unusual", "uncommon, happens now and then",
   "rare, few such cases", "very rare, a record or a first", "a unique case, never seen before"]),
 "memorability": ("Will it be remembered in a week, will you want to retell it?", [
   "forgotten at once", "forgotten by tomorrow", "may be remembered vaguely",
   "remembered for a while, may be retold", "vivid, you will retell it", "etched in memory for years"]),
 "importance": ("Social significance: how much does it change people's lives? A personal story with no effect on others is low even when it is lovely.", [
   "affects nothing beyond the moment", "matters to one person or family", "matters to a small group or a local community",
   "improves life for many people in a city or region", "significant for a country or a large field", "changes the lives of many, historic"]),
 "impact_scale": ("Breadth of reach of the event or its effect.", [
   "one person, one family, one yard", "a street, a school, a small group", "a town or a city district",
   "a whole city or region", "a country", "the whole world"]),
 "usefulness": ("Practical value for the reader: something to visit, apply or use.", [
   "nothing practical", "a curious fact, no use", "a general idea one could use",
   "a concrete tip or a place one can visit", "clear practical advice or a service to use", "a direct step-by-step guide to action"]),
 "clickbait": ("Sensationalism: does the headline promise more than the text delivers?", [
   "honest headline, matches the text", "slightly catchy wording", "a teaser headline, but the text delivers",
   "exaggerated headline, the text gives less", "strongly sensational, the text disappoints", "pure clickbait, the promise is empty"]),
 "controversy": ("Politicisation, polarisation, strife around the topic.", [
   "an uncontroversial topic", "a faint political context", "a topic some people argue about",
   "a clearly debated or political topic", "a polarising issue at the centre", "sharply conflictual, a political fight"]),
 "promo": ("Press release, hidden advertising, promotion of a product, service or company.", [
   "an ordinary news story", "a company or product is named in passing", "a company's achievement reported as news",
   "reads partly like a press release", "mostly promotion of a product or brand", "outright advertising"]),
}
def ladder_questions():
    return {k: {"type": "score", "instructions": PRE + d,
                "criteria": [f"{2*i} — {s}" for i, s in enumerate(steps)]} for k, (d, steps) in L.items()}
```
